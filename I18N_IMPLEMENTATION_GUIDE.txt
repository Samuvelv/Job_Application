BEFORE & AFTER EXAMPLES
=======================

EXAMPLE 1: SIMPLE TEXT IN TEMPLATE
──────────────────────────────────

BEFORE (Hardcoded):
────────────────
<h1 class="h4 fw-bold mb-0">Edit Candidate</h1>

AFTER (With i18n):
─────────────────
<h1 class="h4 fw-bold mb-0" i18n="@@candidate_edit_title">Edit Candidate</h1>

Or using the pipe:
<h1 class="h4 fw-bold mb-0">{{ 'CANDIDATE_EDIT.title' | translate }}</h1>


EXAMPLE 2: BUTTON LABELS
────────────────────────

BEFORE (Hardcoded):
────────────────

AFTER (With i18n):
─────────────────
<button class="btn btn-primary">{{ 'COMMON.save_changes' | translate }}</button>

EXAMPLE 1: ERROR MESSAGES IN SERVICE
────────────────────────────────────

BEFORE (Hardcoded):
────────────────
export class CandidateService {
  saveCandidate(data: any): Observable<any> {
    return this.http.post('/api/candidates', data).pipe(
      catchError(error => {
        this.toast.error('Failed to save candidate');
        return throwError(() => error);
      })
    );
  }
}

AFTER (With i18n):
─────────────────
export class CandidateService {
  constructor(
    private http: HttpClient,
    private toast: ToastService,
    private translate: TranslateService
  ) {}

  saveCandidate(data: any): Observable<any> {
    return this.http.post('/api/candidates', data).pipe(
      catchError(error => {
        this.translate.get('ERROR.save_candidate_failed').subscribe(message => {
          this.toast.error(message);
        });
        return throwError(() => error);
      })
    );
  }
}

    });
  },
  error: (err) => {
    this.translate.get('ERROR.candidate_update_failed').subscribe(msg => {
      this.toast.error(msg);
    });
  }
});

RECOMMENDED PATTERN FOR COMPONENTS:

candidate-edit.component.ts
────────────────────────────

import { Component } from '@angular/core';
import { TranslateModule } from '@ngx-translate/core';

@Component({
  selector: 'app-candidate-edit',
  standalone: true,
  imports: [CommonModule, TranslateModule, ReactiveFormsModule, ...],
  templateUrl: './candidate-edit.component.html',
  styleUrls: ['./candidate-edit.component.scss']
})
export class CandidateEditComponent {
  // Component logic here
  
  constructor(private translate: TranslateService) {}
  
  // For dynamic error messages
  handleError(errorKey: string): void {
    this.translate.get(errorKey).subscribe(message => {
      this.errorMsg = message;
    });
  }
}

candidate-edit.component.html
──────────────────────────────

<div class="container">
  <!-- Use translate pipe for all user-visible strings -->
  <h1>{{ 'CANDIDATE_EDIT.title' | translate }}</h1>
  
  <form [formGroup]="form" (ngSubmit)="onSubmit()">
    
    <!-- Form Labels with i18n -->
    <div class="form-group">
      <label>{{ 'COMMON.first_name' | translate }}<span class="text-danger">*</span></label>
      <input type="text" 
             formControlName="first_name"
             [placeholder]="'COMMON.first_name_placeholder' | translate"
             class="form-control">
      
      <!-- Validation Errors with i18n -->
      @if (form.get('first_name')?.invalid && form.get('first_name')?.touched) {
        <div class="invalid-feedback d-block">
          {{ 'VALIDATION.first_name_required' | translate }}
        </div>
      }
    </div>
    
    <!-- Buttons with i18n -->
    <div class="button-group mt-4">
      <button type="submit" class="btn btn-primary">
        {{ 'COMMON.save_changes' | translate }}
      </button>
      <button type="button" class="btn btn-secondary" (click)="cancel()">
        {{ 'COMMON.cancel' | translate }}
      </button>
    </div>
  </form>
  
  <!-- Error Messages -->
  @if (errorMsg) {
    <div class="alert alert-danger">{{ errorMsg }}</div>
  }
</div>

Unit Test Example:

candidate-edit.component.spec.ts
───────────────────────────────

import { TestBed } from '@angular/core/testing';
import { TranslateModule, TranslateService } from '@ngx-translate/core';
import { CandidateEditComponent } from './candidate-edit.component';

describe('CandidateEditComponent - i18n', () => {
  let component: CandidateEditComponent;
  let fixture: ComponentFixture<CandidateEditComponent>;
  let translate: TranslateService;

  beforeEach(async () => {
    await TestBed.configureTestingModule({
      imports: [
        CandidateEditComponent,
        TranslateModule.forRoot()
      ]
    }).compileComponents();

    fixture = TestBed.createComponent(CandidateEditComponent);
    component = fixture.componentInstance;
    translate = TestBed.inject(TranslateService);
    
    translate.setDefaultLanguage('en');
    translate.use('en');
    fixture.detectChanges();
  });

  it('should display translated button labels', (done) => {
    translate.get('COMMON.save_changes').subscribe(text => {
      expect(fixture.nativeElement.textContent).toContain(text);
      done();
    });
  });

  it('should show translated validation error', () => {
    const control = component.form.get('first_name');
    control?.setErrors({ 'required': true });
    control?.markAsTouched();
    fixture.detectChanges();

    translate.get('VALIDATION.first_name_required').subscribe(text => {
      expect(fixture.nativeElement.textContent).toContain(text);
    });
  });
});

================================================================================
COMMON PITFALLS TO AVOID
================================================================================

1. MISSING TRANSLATION MODULE
   ❌ WRONG:
   import { Component } from '@angular/core';
   export class MyComponent {
     // Missing TranslateModule
   }
   
   ✅ RIGHT:
   import { TranslateModule } from '@ngx-translate/core';
   @Component({
     imports: [TranslateModule, ...],
     // ...
   })


2. NOT HANDLING ASYNC TRANSLATIONS
   ❌ WRONG:
   message: string = this.translate.instant('KEY');
   
   ✅ RIGHT:
   message$ = this.translate.get('KEY');
   // Or use async pipe in template


3. HARDCODING PLACEHOLDERS
   ❌ WRONG:
   <input placeholder="Enter text">
   
   ✅ RIGHT:
   <input [placeholder]="'INPUT.placeholder' | translate">


4. FORGETTING TO EXPORT TRANSLATION MODULE
   ❌ WRONG:
   @Component({
     template: \{{ 'KEY' | translate }}\
     // Missing TranslateModule in imports
   })
   
   ✅ RIGHT:
   @Component({
     imports: [TranslateModule],
     template: \{{ 'KEY' | translate }}\
   })


5. MIXING TRANSLATION METHODS
   ❌ WRONG:
   // Don't mix styles in same component
   {{ 'KEY1' | translate }}
   {{ translationMap.KEY2 }}
   this.translate.instant('KEY3')
   
   ✅ RIGHT:
   // Use consistent approach throughout
